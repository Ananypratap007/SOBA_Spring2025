// Firebase Storage Security Rules
// Copy these to your Firebase Console > Storage > Rules

rules_version = "2";
service firebase.storage {
  match /b/{bucket}/o {
    // Anyone can read images
    match /users/{userId}/profile_images/{imageId} {
      allow read;
      // Only authenticated users can write their own profile images
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    
    // General rule - authenticated users can only access their own data
    match /users/{userId}/{allPaths=**} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
    
    // Default deny
    match /{allPaths=**} {
      allow read, write: if false;
    }
  }
}
